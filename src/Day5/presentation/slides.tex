\begin{frame}[fragile]
	\frametitle{Verkettete Liste}
	\begin{itemize}
		\item Statt eines starren Speicherblocks verbinden wir einzelne \textbf{Knoten} (\texttt{Node}).
		      \begin{itemize}
			      \item Jeder Knoten enthält seine eigene Information (\texttt{value}).
			      \item Jeder Knoten verweist auf einen \textbf{weiteren Knoten} (\texttt{nextNode}).
		      \end{itemize}
		\item Der letzte Knoten verweist auf \texttt{null} (Ende der Kette).
	\end{itemize}
	\vspace{0.8em}
	\centering
	\includegraphics[width=0.88\linewidth]{../exercise/img/linkedlist.png}
\end{frame}

\begin{frame}[fragile]
	\frametitle{ \texttt{Node} und \texttt{List}}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\textbf{Der Knoten (\texttt{Node}):}
			\begin{itemize}
				\item Speichert den Wert (\texttt{value}).
				\item Speichert die Referenz auf den nächsten Knoten (\texttt{nextNode}).
				\item Eine Klasse kann Attribute vom Typ \textbf{ihrer eigenen Klasse} haben!
			\end{itemize}
		\end{column}
		\pause
		\begin{column}{0.5\textwidth}
			\textbf{Die Liste (\texttt{List}):}
			\begin{itemize}
				\item Muss sich nur merken, wo der Anfang ist: \textbf{Startknoten} (\texttt{startNode}).
				\item Ist \texttt{startNode == null}, ist die Liste leer.
			\end{itemize}
		\end{column}
	\end{columns}
	\pause
	\vspace{1.2em}
	\begin{tcolorbox}[colback=accent!10, colframe=accent]
		Alle Elemente der Liste erreichen wir, indem wir vom Startknoten aus Schritt für Schritt den Pfeilen folgen!
	\end{tcolorbox}
\end{frame}

\begin{frame}[fragile]
	\frametitle{Durch die Kette wandern}
	\begin{itemize}
		\item Wie schaut man sich alle Elemente nacheinander an (z.\,B. zum Ausgeben)?
		\item \textbf{Das Vorgehen:}
		      \begin{enumerate}
			      \item Starte beim ersten Knoten (\texttt{startNode}).
			      \item Schau dir den Wert des aktuellen Knotens an.
			      \item Gehe einen Schritt weiter zum Nachfolger (\texttt{nextNode}).
			      \item Wiederhole das, bis du bei \texttt{null} (dem Ende) ankommst!
		      \end{enumerate}
	\end{itemize}

\begin{frame}[fragile]
	\frametitle{Rekursion auf Datenstrukturen}
	\begin{itemize}
		\item Eine Liste ist von Natur aus rekursiv aufgebaut:
		      \begin{center}
			      \textit{„Ein Knoten, gefolgt von einer etwas kleineren Rest-Liste.“}
		      \end{center}
		\item\pause \textbf{Das Prinzip der Delegation:}
		      \begin{itemize}
			      \item Ein Knoten erledigt seinen eigenen Anteil (z.\,B. seinen eigenen Wert hinzuzählen).
			      \item Dann delegiert er an seinen Nachfolger: \textit{„Mach du bitte mit dem Zwischenergebnis weiter!“}
		      \end{itemize}
		\item\pause \textbf{Basisfall (Abbruch):}
		      \begin{itemize}
			      \item Wenn ein Knoten keinen Nachfolger mehr hat (\texttt{nextNode == null}), ist das Ende der Kette erreicht und das Gesamtergebnis steht fest!
		      \end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}[fragile]
	\frametitle{Bäume (Trees)}
	\begin{itemize}
		\item In einer verketteten Liste hat jeder Knoten maximal \textbf{einen} Nachfolger (linear).
		\item Was passiert, wenn ein Knoten \textbf{zwei oder mehr} Nachfolger haben kann?
		\item $\to$ Es entsteht eine verzweigte Hierarchie: Ein \textbf{Baum}!
	\end{itemize}
	\pause
	\vspace{0.3em}
	\begin{columns}
		\begin{column}{0.5\textwidth}
			\begin{itemize}
				\item \textbf{Wurzel (Root):} Oberster Startknoten
				\item \textbf{Kinder (Children):} Untergeordnete Knoten
				\item \textbf{Blätter (Leaves):} Knoten ohne Kinder
				\item \textbf{Binärbaum:} Maximal \textbf{zwei} Kinder pro Knoten (\texttt{links} und \texttt{rechts})
			\end{itemize}
		\end{column}
		\begin{column}{0.5\textwidth}
			\centering
			\includegraphics[width=0.85\linewidth]{../exercise/img/baum.png}
		\end{column}
	\end{columns}
\end{frame}

\begin{frame}[fragile]
	\frametitle{Warum Bäume?}
	\begin{itemize}
		\item Bäume spiegeln viele reale Strukturen wider (Dateisysteme mit Ordnern, Stammbäume, HTML-Dokumente).
		\item \textbf{Extrem schnelle Suche (Binäre Suchbäume):}
		      \begin{itemize}
			      \item Bei jedem Schritt entscheidet man: Geh nach links oder nach rechts?
			      \item Wie beim Zahlenraten: „Ist die Zahl größer oder kleiner?“
			      \item Mit jedem Schritt halbiert sich der verbleibende Suchraum!
		      \end{itemize}
		\item Mehr dazu und spannende Algorithmen findet ihr in der \textbf{Highperformer-Aufgabe} auf dem Übungsblatt.
	\end{itemize}
\end{frame}
